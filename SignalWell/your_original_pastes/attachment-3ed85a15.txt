# training, validation, test, and real-world inference (Cells 8, 16, 17 all
# share this same `eval_tf`/`norm_tf`).

# ---- Cell 7 — MTCNN preprocessing ----
mtcnn = MTCNN(image_size=CONFIG["img_size"], margin=CONFIG["mtcnn_margin"],
              post_process=False, select_largest=True, device=device)

def mtcnn_crop(pil_img):
    """Detect + align + crop a single face. Returns a raw (unnormalized) 160x160
    PIL RGB image, or None if no face was detected."""
    pil_img = pil_img.convert("RGB")
    with torch.no_grad():
        face_tensor = mtcnn(pil_img)
    if face_tensor is None:
        return None
    face_tensor = face_tensor.clamp(0, 255).byte()
    return T.ToPILImage()(face_tensor)

def precrop_split(hf_split, name):
    crops, labels, dropped = [], [], 0
    for i in range(len(hf_split)):
        item = hf_split[i]
        face = mtcnn_crop(item["image"])
        if face is None:
            dropped += 1
            continue
        crops.append(face)
        labels.append(item["label"])
    print(f"[{name}] kept {len(crops)}/{len(hf_split)} images "
          f"({dropped} had no face detected by MTCNN and were dropped)")
    return crops, labels

print("Running MTCNN once over each split (this can take a few minutes)...")

# ---- Cell 8 — Dataset and DataLoaders ----
IMG_SIZE = CONFIG["img_size"]
norm_tf = T.Normalize(CONFIG["norm_mean"], CONFIG["norm_std"])

train_aug = T.Compose([
    T.RandomHorizontalFlip(p=0.5),
    T.RandomRotation(degrees=8),
    T.RandomAffine(degrees=0, translate=(0.05, 0.05)),
    T.ColorJitter(brightness=0.2, contrast=0.2, saturation=0.1, hue=0.02),
    T.RandomApply([T.GaussianBlur(kernel_size=3)], p=0.15),
    T.ToTensor(),
    norm_tf,
])

eval_tf = T.Compose([
    T.ToTensor(),
    norm_tf,
])

class FaceCropDataset(Dataset):
    def __init__(self, crops, labels, train=False):
        self.crops = crops
        self.labels = labels
        self.train = train
    def __len__(self):
        return len(self.crops)
    def __getitem__(self, idx):
        face = self.crops[idx]
        tf = train_aug if self.train else eval_tf
        return tf(face), self.labels[idx]

train_ds = FaceCropDataset(train_crops, train_crop_labels, train=True)
val_ds = FaceCropDataset(val_crops, val_crop_labels, train=False)

# ---- Cell 10 — Build v7 model ----
class FaceEmotionModel(nn.Module):
    def __init__(self, num_classes, dropout=0.3):
        super().__init__()
        self.backbone = InceptionResnetV1(pretrained="vggface2", classify=False)
        self.dropout = nn.Dropout(dropout)
        self.fc = nn.Linear(512, num_classes)

    def forward(self, x):
        emb = self.backbone(x)
        emb = self.dropout(emb)
        return self.fc(emb)

model = FaceEmotionModel(NUM_CLASSES, dropout=CONFIG["dropout"]).to(device)
n_params = sum(p.numel() for p in model.parameters())
print(f"Total parameters: {n_params:,}")



# ---- Cell 11 — Training functions ----
criterion = nn.CrossEntropyLoss(weight=class_weights_t)
scaler = torch.cuda.amp.GradScaler(enabled=torch.cuda.is_available())
best_val_acc = 0.0

@torch.no_grad()
def evaluate(model, loader):
    model.eval()
    total, correct, loss_sum = 0, 0, 0.0
    all_preds, all_labels = [], []
    for x, y in loader:
        x, y = x.to(device), y.to(device)
        with torch.cuda.amp.autocast(enabled=torch.cuda.is_available()):
            out = model(x)
            loss = criterion(out, y)
        loss_sum += loss.item() * x.size(0)
        preds = out.argmax(1)
        correct += (preds == y).sum().item()
        total += x.size(0)
        all_preds.append(preds.cpu().numpy())
        all_labels.append(y.cpu().numpy())
    return correct / total, loss_sum / total, np.concatenate(all_preds), np.concatenate(all_labels)

def train_one_phase(model, loader, val_loader, optimizer, scheduler, n_epochs, phase_name,
                     patience=CONFIG["early_stop_patience"]):
    global best_val_acc
    epochs_no_improve = 0
    for epoch in range(n_epochs):
        model.train()
        total, correct, running_loss = 0, 0, 0.0
        for x, y in loader:
            x, y = x.to(device), y.to(device)
            optimizer.zero_grad()
            with torch.cuda.amp.autocast(enabled=torch.cuda.is_available()):
                out = model(x)
                loss = criterion(out, y)
            scaler.scale(loss).backward()
            scaler.step(optimizer)
            scaler.update()
            if scheduler is not None:
                scheduler.step()
            running_loss += loss.item() * x.size(0)
            correct += (out.argmax(1) == y).sum().item()
            total += x.size(0)

        val_acc, val_loss, _, _ = evaluate(model, val_loader)
        train_acc = correct / total
        print(f"[{phase_name}] epoch {epoch+1}/{n_epochs} "
              f"train_loss={running_loss/total:.4f} train_acc={train_acc:.4f} "
              f"val_loss={val_loss:.4f} val_acc={val_acc:.4f} best_val_acc={best_val_acc:.4f}")

        if val_acc > best_val_acc:
            best_val_acc = val_acc
            epochs_no_improve = 0
            torch.save({
                "state_dict": model.state_dict(),
                "labels": label_names,
                "class_to_idx": {l: i for i, l in enumerate(label_names)},
                "val_acc": best_val_acc,
                "preprocessing": {
                    "img_size": CONFIG["img_size"],
                    "norm_mean": CONFIG["norm_mean"],
                    "norm_std": CONFIG["norm_std"],
                    "mtcnn_margin": CONFIG["mtcnn_margin"],
                    "note": "Single normalization after a raw (post_process=False) MTCNN crop",
                },
                "split_is_subject_independent": SPLIT_IS_SUBJECT_INDEPENDENT,
            }, SAVE_PATH)
        else:
            epochs_no_improve += 1
            if epochs_no_improve >= patience:
                print(f"Early stopping in {phase_name} (no val improvement for {patience} epochs)")
                break



# ---- Cell 12 — Train model ----
print("\n--- Phase 1: frozen backbone, train head only ---")
for p in model.backbone.parameters():
    p.requires_grad = False
opt1 = torch.optim.AdamW([p for p in model.parameters() if p.requires_grad],
                          lr=CONFIG["phase1_lr"], weight_decay=CONFIG["weight_decay"])
sched1 = torch.optim.lr_scheduler.OneCycleLR(opt1, max_lr=CONFIG["phase1_lr"],
    steps_per_epoch=len(train_loader), epochs=CONFIG["phase1_epochs"])
train_one_phase(model, train_loader, val_loader, opt1, sched1, CONFIG["phase1_epochs"], "phase1-head")

print("\n--- Phase 2: unfreeze later backbone blocks ---")
for name, p in model.backbone.named_parameters():
    if any(k in name for k in ["block8", "repeat_3", "last_linear", "last_bn"]):
        p.requires_grad = True
opt2 = torch.optim.AdamW([p for p in model.parameters() if p.requires_grad],
                          lr=CONFIG["phase2_lr"], weight_decay=CONFIG["weight_decay"])
sched2 = torch.optim.lr_scheduler.OneCycleLR(opt2, max_lr=CONFIG["phase2_lr"],
    steps_per_epoch=len(train_loader), epochs=CONFIG["phase2_epochs"])
train_one_phase(model, train_loader, val_loader, opt2, sched2, CONFIG["phase2_epochs"], "phase2-partial")

print("\n--- Phase 3 (optional): full fine-tune, small LR ---")
for p in model.parameters():
    p.requires_grad = True
opt3 = torch.optim.AdamW(model.parameters(), lr=CONFIG["phase3_lr"], weight_decay=CONFIG["weight_decay"])
sched3 = torch.optim.lr_scheduler.CosineAnnealingLR(opt3, T_max=CONFIG["phase3_epochs"] * len(train_loader))
train_one_phase(model, train_loader, val_loader, opt3, sched3, CONFIG["phase3_epochs"], "phase3-full")

print(f"\nBest validation accuracy: {best_val_acc:.4f}")
print(f"Best checkpoint saved to: {SAVE_PATH}")

# ---- Cell 16 — Real-world image testing ----
mtcnn_infer = MTCNN(image_size=CONFIG["img_size"], margin=CONFIG["mtcnn_margin"],
                     post_process=False, keep_all=True, device=device)

def predict_emotions(image_path_or_pil, show=True):
    """
    1. load image
    2. detect face(s) with MTCNN
    3. align/crop each to 160x160
    4. apply the SAME normalization used in training (eval_tf, no extra steps)
    5. predict all 7 emotion probabilities per face
    6. return top prediction + confidence per face (and optionally display them)
    """
    if isinstance(image_path_or_pil, str):
        img = Image.open(image_path_or_pil).convert("RGB")
    else:
        img = image_path_or_pil.convert("RGB")

    with torch.no_grad():
        faces = mtcnn_infer(img)

    if faces is None:
        print("No face detected.")
        return []
    if faces.dim() == 3:
        faces = faces.unsqueeze(0)

    results = []
    model.eval()
    with torch.no_grad():
        for i in range(faces.shape[0]):
            face_raw = faces[i].clamp(0, 255).byte()
            face_pil = T.ToPILImage()(face_raw)
            x = eval_tf(face_pil).unsqueeze(0).to(device)
            logits = model(x)
            probs = F.softmax(logits, dim=1)[0].cpu().numpy()
            top_idx = int(probs.argmax())
            results.append({
                "face_index": i,
                "top_emotion": label_names[top_idx],
                "confidence": float(probs[top_idx]),
                "all_probabilities": dict(zip(label_names, probs.tolist())),
                "face_image": face_pil,
            })

    if show:
        fig, axes = plt.subplots(1, len(results), figsize=(4 * len(results), 4))
        if len(results) == 1:
            axes = [axes]
        for ax, r in zip(axes, results):
            ax.imshow(r["face_image"])
            ax.set_title(f'{r["top_emotion"]} ({r["confidence"]*100:.1f}%)')
            ax.axis("off")
        plt.tight_layout()
        plt.show()

    return results

# ---- Cell 17 — PDF → 300 DPI → face detection pipeline ----
from pdf2image import convert_from_path

def predict_emotions_from_pdf(pdf_path, dpi=300, show=True):
    pages = convert_from_path(pdf_path, dpi=dpi)
    print(f"Rendered {len(pages)} page(s) from {pdf_path} at {dpi} DPI")

    all_results = []
    for page_num, page_img in enumerate(pages):
        print(f"\n--- Page {page_num + 1} ---")
        page_results = predict_emotions(page_img, show=show)
        for r in page_results:
            r["page"] = page_num + 1
        all_results.append(page_results)
    return all_results
